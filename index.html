GTA Clipper packages

Fedora (dnf)

sudo curl -fsSL -o /etc/yum.repos.d/gta-clipper.repo https://usingblender.github.io/live-clipping-software/rpm/gta-clipper.repo
sudo dnf install gta-clipper
systemctl --user enable --now gta-clipper

Ubuntu / Debian (apt)

sudo install -d /etc/apt/keyrings
curl -fsSL https://usingblender.github.io/live-clipping-software/gpg.key | sudo gpg --dearmor -o /etc/apt/keyrings/gta-clipper.gpg
echo "deb [signed-by=/etc/apt/keyrings/gta-clipper.gpg] https://usingblender.github.io/live-clipping-software/deb stable main" | sudo tee /etc/apt/sources.list.d/gta-clipper.list
sudo apt update && sudo apt install gta-clipper
systemctl --user enable --now gta-clipper
