Live Clipper packages

Fedora (dnf)

sudo curl -fsSL -o /etc/yum.repos.d/live-clipper.repo https://usingblender.github.io/live-clipping-software/rpm/live-clipper.repo
sudo dnf install live-clipper
systemctl --user enable --now live-clipper

Ubuntu / Debian (apt)

sudo install -d /etc/apt/keyrings
curl -fsSL https://usingblender.github.io/live-clipping-software/gpg.key | sudo gpg --dearmor -o /etc/apt/keyrings/live-clipper.gpg
echo "deb [signed-by=/etc/apt/keyrings/live-clipper.gpg] https://usingblender.github.io/live-clipping-software/deb stable main" | sudo tee /etc/apt/sources.list.d/live-clipper.list
sudo apt update && sudo apt install live-clipper
systemctl --user enable --now live-clipper
